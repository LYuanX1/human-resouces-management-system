USE stu_db;

DROP TABLE IF EXISTS student;

CREATE TABLE student (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '干员编号',
    name VARCHAR(50) NOT NULL COMMENT '代号/姓名',
    age INT NOT NULL COMMENT '年龄',
    gender VARCHAR(10) NOT NULL COMMENT '性别',
    major VARCHAR(100) DEFAULT '特勤组' COMMENT '专业/技能',
    department VARCHAR(50) DEFAULT '突击' COMMENT '部门/兵种',
    avatar VARCHAR(255) DEFAULT 'assets/avatar_default.png' COMMENT '头像路径'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
